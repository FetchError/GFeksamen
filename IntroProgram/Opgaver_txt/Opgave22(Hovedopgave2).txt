namespace IntroProgram;

class Program
{
    static void Main(string[] args)
    {   // Byder brugeren velkommen
        Console.WriteLine("Velkommen til Kino Biograf!");
        // Spørger brugeren om deres alder
        Console.Write("Indtast din alder: ");

        // Konverterer brugerens input til et heltal
        int alder = Convert.ToInt32(Console.ReadLine());

        // Tjekker om brugeren har en negativ alder 
        if (alder < 0)
        {   // Hvis alderen er negativ, udskrives en besked til brugeren at den ikke må være det
            Console.WriteLine("Du kan ikke have en negativ alder");
        } //Tjekker om brugeren er under 18
        else if (alder < 17)
        {   // Hvis alderen er under 18, udskrives denne besked til brugeren
            Console.WriteLine("Pris på billet: 45 DKK");
        }   // Tjekker om brugeren er under 66
        else if (alder < 66)
        {   // Hvis alderen er under 66, udskrives denne besked til brugeren
            Console.WriteLine("Pris på billet: 75 DKK");
        } else 
        {   // Hvis alderen er over 66, udskrives denne besked til brugeren
            Console.WriteLine("Pris på billet: 35 DKK");
        }


        Console.ReadKey();
    }
}
