namespace IntroProgram;

class Program
{
    static void Main(string[] args)
    {
        // Sprøger bruger om input i form af tal som brugeren vil se tabellen på
        Console.Write("Indtast et tal for at se tabellen: ");

        // Konverterer brugerens input til et heltal og gemmer det
        int tal = Convert.ToInt32(Console.ReadLine());

        // Laver en for-løkke der kører fra 1 til 10, og udskriver tabellen som brugerens input er
        for (int i = 1; i <= 10; i++)
        {
            // Her udskrives tabellen som brugeren spørger efter
            Console.WriteLine("{0}   *   {1}   =   {2}", tal, i, tal * i);
        }

        Console.ReadKey();
    }
}
