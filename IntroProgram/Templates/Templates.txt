// ---- 5. OPERATORER ----

if (a != b) { }   // forskellig fra
if (a < b)  { }   // mindre end
if (a > b)  { }   // større end

// ---- 9. IF-ELSE ----

{

{

// ---- 12. DO-WHILE ----
Random random = new Random();
int tal;
do
{
    tal = random.Next(1, 10); // Tilfældigt tal 1-9
    Console.WriteLine("Tal: " + tal);
} while (tal != 5); // Fortsæt indtil vi får 5

// ---- 13. SWITCH-CASE ----

switch (dag)
{
    case 1:
        Console.WriteLine("Mandag");

    case 2:

        break;
    case 3:
