namespace IntroProgram;

class Program
{
    static void Main(string[] args)
    {
        // Initialisering af variabler på billet priserne
        int voksen_billet = 150;
        int børne_billet = 100;
        int barn_m_sodavand_billet = 110;

        // Deklaration af variabler på antal biletter
        int voksen_antal = 0;
        int børn_antal = 0;
        int barn_m_s_antal = 0;

        // Initialisering af variabler på total og antal billetter
        int total = 0;
        int antal = 0;
        // Deklaration af variabler på valget af billet type
        string billettype_valg;
        // Variabel til do while løkke for at fortsætte eller købe flere billetter
        string fortsæt = "j";

        // Byder brugeren velkommen
        Console.WriteLine("=======================================");
        Console.WriteLine("Velkommen til Natur og Forlystelsespark");
        Console.WriteLine("=======================================");

        do
        {
            // Udskriver billet menu
            Console.WriteLine(" ");
            Console.WriteLine("Vælg billettype:");
            Console.WriteLine(" ");
            Console.WriteLine($"(V) Voksen billet               {voksen_billet} kr");
            Console.WriteLine($"(B) Børne billet                {børne_billet} kr");
            Console.WriteLine($"(BS) Børne billet med Sodavand  {barn_m_sodavand_billet} kr");
            Console.WriteLine(" ");
            Console.WriteLine("Hvilken billettype ønsker du?");

            billettype_valg = (Console.ReadLine() ?? "").ToLower(); // Gemmer brugerens valg og sikrer mod null exception 

            if (billettype_valg == "v") // Tjekker for om brugeren taster "v"
            {
                Console.WriteLine($"Voksen billet - {voksen_billet} kr"); // Viser type og pris på brugerens valg
                Console.Write("Hvor mange billetter vil du købe? "); // Spørger brugeren om antal billetter
                antal = int.Parse(Console.ReadLine() ?? ""); // // Brugeren indtaster antal som bliver gemt

                if (antal < 0) // Tjekker om antallet er negativt
                {
                    Console.WriteLine("Du kan ikke købe negativt antal billetter"); // Udskriver hvis det er negativt
                }
                else if (voksen_antal + antal > 10) // Tjekker om grænsen på 10 billetter bliver overskredet
                {
                    Console.WriteLine("Du kan max købe 10 voksenbilletter"); // Udskriver hvis antal er over 10
                }
                else
                {
                    total += voksen_billet * antal; // Lægger antal voksen billetter til total
                    voksen_antal += antal; // Lægger antal voksenbilletter til antal
                }
            }
            else if (billettype_valg == "b") // Tjekker for om brugeren taster "b"
            {
                // Hvis brugeren taster "b", bliver de sprugt om antal biletter de vil købe
                Console.WriteLine($"Børne billet - {børne_billet} kr"); // Viser type og pris på brugerens valg
                Console.Write("Hvor mange billetter vil du købe? "); // Spørger brugeren om antal billetter
                antal = int.Parse(Console.ReadLine() ?? ""); // Brugeren indtaster antal som bliver gemt

                if (antal < 0) // Tjekker om antallet er negativt
                {
                    Console.WriteLine("Du kan ikke købe negativt antal billetter"); // Udskriver hvis det er negativt
                }
                else if (børn_antal + antal > 10) // Tjekker for om grænsen på de 10 billeter pr person
                {
                    Console.WriteLine("Du kan max købe 10 børnebilletter"); // Udskriver hvis antal er over 10
                }
                else
                {
                    total += børne_billet * antal; // Lægger antal børne billetter til total
                    børn_antal += antal; // Lægger antal børnebilletter til antal
                }
            }
            else if (billettype_valg == "bs") // Tjekker for om brugeren taster "bs"
            {
                Console.WriteLine($"Barn med sodavand - {barn_m_sodavand_billet} kr"); // Viser type og pris på brugerens valg
                Console.Write("Hvor mange billetter vil du købe? ");// Spørger brugeren om antal billetter
                antal = int.Parse(Console.ReadLine() ?? ""); // Brugeren indtaster antal som bliver gemt

                if (antal < 0) // Tjekker om antallet er negativt
                {
                    Console.WriteLine("Du kan ikke købe negativt antal billetter"); // Udskriver hvis det er negativt
                }
                else if (barn_m_s_antal + antal > 10) // Tjekker for om grænsen på de 10 billeter pr person
                {
                    Console.WriteLine("Du kan max købe 10 børnebilletter med sodavand"); // Udskriver hvis antal er over 10
                }
                else
                {
                    total += barn_m_sodavand_billet * antal; // Lægger antal børne billetter med sodavand til total
                    barn_m_s_antal += antal; // Lægger antal børnebilletter med sodavand til antal
                }
            }
            else
            {   // Udskriver denne fejlbesked hvis brugeren hverken taster v, b eller bs
                Console.WriteLine("Den valgte billet findes ikke, prøv igen"); 
            }

            Console.Write("Vil du købe flere billetter? (j/n): "); // Spørger om brugeren vil købe flere eller afregne
            fortsæt = (Console.ReadLine() ?? "n").ToLower(); // Gemmer svaret og sikrer mod null exception

        } while (fortsæt == "j"); // Kører løkken igen hvis brugeren indtaster "j"

        

        // Summen af alle købte billetter i DKK
        int total_dkk = total;
        // Konverterer DKK til EUR
        double total_eur = total_dkk / 7.4617;

        // Udskriver den fulde bestilling med antal og type billetter
        Console.WriteLine("Tak for dit køb hos Natur og Forlystelsespark");
        Console.WriteLine($"Du har købt: {voksen_antal} voksenbilletter, {børn_antal} børnebilletter og {barn_m_s_antal} børnebilletter med sodavand");
        Console.WriteLine($"Totalpris: {total_dkk} DKK");
        Console.WriteLine($"Totalpris i Euro: {total_eur:N2} EUR"); // Udkskrikver prisen i EURO med 2 dicimaler 


        Console.ReadLine();
    }
}
